```python
def longest_palindrome(s: str) -> str:
    """Return the longest palindromic substring using Manacher's algorithm in O(n)."""
    if len(s) <= 1:
        return s

    # Transform: insert '#' between every character to unify odd/even palindromes
    # e.g. "abc" -> "#a#b#c#"
    t: list[str] = []
    for ch in s:
        t.append('#')
        t.append(ch)
    t.append('#')
    n: int = len(t)

    # d[i] = radius (half-length) of the palindrome centered at i in t
    d: list[int] = [0] * n
    center: int = 0
    right: int = 0  # rightmost boundary of the known palindrome

    for i in range(n):
        # If i is inside the current right boundary, use mirror to skip work
        if i < right:
            mirror: int = 2 * center - i
            d[i] = min(d[mirror], right - i)
        else:
            d[i] = 0

        # Expand outward while characters match
        while (
            i + d[i] + 1 < n
            and i - d[i] - 1 >= 0
            and t[i + d[i] + 1] == t[i - d[i] - 1]
        ):
            d[i] += 1

        # Update the rightmost boundary if this palindrome extends further
        if i + d[i] > right:
            center = i
            right = i + d[i]

    # Find the center with the largest radius
    max_radius: int = 0
    max_center: int = 0
    for i in range(n):
        if d[i] > max_radius:
            max_radius = d[i]
            max_center = i

    # Extract the palindrome from the transformed string
    start: int = max_center - max_radius
    end: int = max_center + max_radius
    result: str = ''.join(ch for ch in t[start:end + 1] if ch != '#')

    return result
```
