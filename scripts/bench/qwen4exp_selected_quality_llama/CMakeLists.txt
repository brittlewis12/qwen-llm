cmake_minimum_required(VERSION 3.19)
project(qwen4exp_selected_quality_llama LANGUAGES C CXX)

set(QWEN4EXP_LLAMA_COMMIT "6c84c7d5d8833c6e0df69628f75a0f599797934e")
set(LLAMA_CPP_SOURCE "" CACHE PATH "Pinned llama.cpp source checkout")
set(QWEN4EXP_LLAMA_TREE "" CACHE STRING "Pinned llama.cpp tree")
set(QWEN4EXP_LLAMA_EXPORT_MANIFEST_SHA256 "" CACHE STRING "Read-only export manifest")
set(QWEN4EXP_BUILD_POLICY_SHA256 "" CACHE STRING "Wrapper-owned build policy")
if(NOT LLAMA_CPP_SOURCE)
    message(FATAL_ERROR "LLAMA_CPP_SOURCE is required")
endif()
string(LENGTH "${QWEN4EXP_LLAMA_TREE}" QWEN4EXP_LLAMA_TREE_LENGTH)
if((NOT QWEN4EXP_LLAMA_TREE_LENGTH EQUAL 40 AND NOT QWEN4EXP_LLAMA_TREE_LENGTH EQUAL 64)
   OR NOT QWEN4EXP_LLAMA_TREE MATCHES "^[0-9a-f]+$")
    message(FATAL_ERROR "QWEN4EXP_LLAMA_TREE must be a lowercase Git tree ID")
endif()
foreach(REQUIRED_SHA QWEN4EXP_LLAMA_EXPORT_MANIFEST_SHA256 QWEN4EXP_BUILD_POLICY_SHA256)
    string(LENGTH "${${REQUIRED_SHA}}" REQUIRED_SHA_LENGTH)
    if(NOT REQUIRED_SHA_LENGTH EQUAL 64 OR NOT "${${REQUIRED_SHA}}" MATCHES "^[0-9a-f]+$")
        message(FATAL_ERROR "${REQUIRED_SHA} must be a lowercase SHA-256")
    endif()
endforeach()

set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_UI OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(LLAMA_SUBPROCESS OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(GGML_CCACHE OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(GGML_METAL ON CACHE BOOL "" FORCE)
set(GGML_METAL_EMBED_LIBRARY ON CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)

add_subdirectory("${LLAMA_CPP_SOURCE}" llama.cpp EXCLUDE_FROM_ALL)

set_property(
    DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/main.cpp"
)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/main.cpp" QWEN4EXP_RUNNER_MAIN_SHA256)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/CMakeLists.txt" QWEN4EXP_RUNNER_CMAKE_SHA256)
set(QWEN4EXP_RUNNER_SOURCE_DOMAIN
    "qwen4exp-selected-quality-llama-core-sources-v1\nmain.cpp=${QWEN4EXP_RUNNER_MAIN_SHA256}\nCMakeLists.txt=${QWEN4EXP_RUNNER_CMAKE_SHA256}\n"
)
string(SHA256 QWEN4EXP_RUNNER_SOURCE_MANIFEST_SHA256 "${QWEN4EXP_RUNNER_SOURCE_DOMAIN}")
string(TOUPPER "${CMAKE_BUILD_TYPE}" QWEN4EXP_BUILD_TYPE_UPPER)
set(QWEN4EXP_RUNNER_FLAGS
    "${CMAKE_CXX_FLAGS}|${CMAKE_CXX_FLAGS_${QWEN4EXP_BUILD_TYPE_UPPER}}"
)
string(SHA256 QWEN4EXP_RUNNER_FLAGS_SHA256 "${QWEN4EXP_RUNNER_FLAGS}")

add_executable(qwen4exp-selected-quality-llama
    main.cpp
    "${LLAMA_CPP_SOURCE}/vendor/hash/sha256/sha256.c"
)
target_compile_features(qwen4exp-selected-quality-llama PRIVATE cxx_std_17)
target_compile_definitions(qwen4exp-selected-quality-llama PRIVATE
    QWEN4EXP_LLAMA_COMMIT="${QWEN4EXP_LLAMA_COMMIT}"
    QWEN4EXP_LLAMA_TREE="${QWEN4EXP_LLAMA_TREE}"
    QWEN4EXP_LLAMA_EXPORT_MANIFEST_SHA256="${QWEN4EXP_LLAMA_EXPORT_MANIFEST_SHA256}"
    QWEN4EXP_BUILD_POLICY_SHA256="${QWEN4EXP_BUILD_POLICY_SHA256}"
    QWEN4EXP_RUNNER_MAIN_SHA256="${QWEN4EXP_RUNNER_MAIN_SHA256}"
    QWEN4EXP_RUNNER_CMAKE_SHA256="${QWEN4EXP_RUNNER_CMAKE_SHA256}"
    QWEN4EXP_RUNNER_SOURCE_MANIFEST_SHA256="${QWEN4EXP_RUNNER_SOURCE_MANIFEST_SHA256}"
    QWEN4EXP_RUNNER_COMPILER_ID="${CMAKE_CXX_COMPILER_ID}"
    QWEN4EXP_RUNNER_COMPILER_VERSION="${CMAKE_CXX_COMPILER_VERSION}"
    QWEN4EXP_RUNNER_BUILD_TYPE="${CMAKE_BUILD_TYPE}"
    QWEN4EXP_RUNNER_FLAGS_SHA256="${QWEN4EXP_RUNNER_FLAGS_SHA256}"
)
target_include_directories(qwen4exp-selected-quality-llama PRIVATE
    "${LLAMA_CPP_SOURCE}/include"
    "${LLAMA_CPP_SOURCE}/ggml/include"
    "${LLAMA_CPP_SOURCE}/vendor"
    "${LLAMA_CPP_SOURCE}/vendor/hash"
    "${LLAMA_CPP_SOURCE}/vendor/hash/sha256"
)
target_link_libraries(qwen4exp-selected-quality-llama PRIVATE llama)
