cmake_minimum_required(VERSION 3.20)
project(k2_checkpoint_oracle LANGUAGES C CXX)

set(K2_REFERENCE_REVISION "42adf019f76013dac873b5b43950d54d5ab27216")
if(NOT LLAMA_CPP_DIR)
    message(FATAL_ERROR "Set LLAMA_CPP_DIR to the pinned IFM K2 fork")
endif()
find_package(Git REQUIRED)
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${LLAMA_CPP_DIR}" rev-parse HEAD
    OUTPUT_VARIABLE reference_head OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${LLAMA_CPP_DIR}" status --porcelain --untracked-files=all
    OUTPUT_VARIABLE reference_dirty OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT reference_head STREQUAL K2_REFERENCE_REVISION OR NOT reference_dirty STREQUAL "")
    message(FATAL_ERROR "Oracle requires the clean pinned IFM source revision")
endif()

set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(GGML_METAL ON CACHE BOOL "" FORCE)
set(GGML_METAL_EMBED_LIBRARY ON CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
add_subdirectory("${LLAMA_CPP_DIR}" "${CMAKE_BINARY_DIR}/llama")
add_executable(k2_checkpoint_oracle main.cpp)
target_compile_features(k2_checkpoint_oracle PRIVATE cxx_std_17)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/main.cpp")
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/main.cpp" K2_WRAPPER_SHA256)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/CMakeLists.txt" K2_CMAKE_SHA256)
target_compile_definitions(k2_checkpoint_oracle PRIVATE
    K2_REFERENCE_REVISION="${K2_REFERENCE_REVISION}"
    K2_WRAPPER_SHA256="${K2_WRAPPER_SHA256}"
    K2_CMAKE_SHA256="${K2_CMAKE_SHA256}")
target_link_libraries(k2_checkpoint_oracle PRIVATE llama)
set_target_properties(k2_checkpoint_oracle PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
